☁️ تكامل السحابة وإدارة التراخيص عبر Firebase Firestore

يتيح تطبيق SamMikrotik للمستخدمين الذين سجلوا دخولهم عبر Google مزامنة التفعيل السحابي مباشرة عند الاتصال بالإنترنت عبر المستند التالي في Firestore:

// مسار مستند الاشتراك في Firestore: // Path: users/{uid}/subscription/status { "tier": "PRO", // الخطة: PRO أو FREE "expirationEpochMs": 1789900000000, // تاريخ الانتهاء بالمللي ثانية (-1 أو null للدائم) "customerName": "شبكة الأمل السريعة", "updatedAt": 1730000000000, "hardwareDeviceId": "SAM-A1B2-C3D4-E5F6" }

كود تفعيل العميل برمجياً (Node.js / Firebase Admin SDK):

const admin = require("firebase-admin"); const db = admin.firestore(); async function activateUserSubscription(uid, months = 12, customerName = "") { const now = Date.now(); const expiry = months > 0 ? now + (months * 30 * 24 * 60 * 60 * 1000) : -1; await db.doc(`users/${uid}/subscription/status`).set({ tier: "PRO", expirationEpochMs: expiry, customerName: customerName, updatedAt: now }, { merge: true }); console.log(`User ${uid} successfully activated to PRO for ${months} months.`); }
